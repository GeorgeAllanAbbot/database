cmake_minimum_required(VERSION 3.16)
project(database_file_manager LANGUAGES CXX)
add_library(file_manage file_manage/file_manage.cpp file_manage/buf_page_manager.cpp)
target_include_directories(file_manage PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(file_manage PUBLIC cxx_std_17)
target_compile_options(file_manage PRIVATE -Wall -Wextra -Wpedantic)
add_library(record_manage record_manage/schema.cpp record_manage/record_manager.cpp)
target_link_libraries(record_manage PUBLIC file_manage)
target_compile_options(record_manage PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if (BUILD_TESTING)
    add_executable(record_manage_test tests/record_manage_test.cpp)
    target_link_libraries(record_manage_test PRIVATE record_manage)
    target_compile_options(record_manage_test PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME record_manage_test COMMAND record_manage_test)
    add_test(NAME record_persistence_write COMMAND record_manage_test seed "${CMAKE_CURRENT_BINARY_DIR}/record-persistence")
    set_tests_properties(record_persistence_write PROPERTIES FIXTURES_SETUP record_persistence)
    add_test(NAME record_persistence_read COMMAND record_manage_test verify "${CMAKE_CURRENT_BINARY_DIR}/record-persistence")
    set_tests_properties(record_persistence_read PROPERTIES FIXTURES_REQUIRED record_persistence)
    add_test(NAME record_persistence_cleanup COMMAND record_manage_test cleanup "${CMAKE_CURRENT_BINARY_DIR}/record-persistence")
    set_tests_properties(record_persistence_cleanup PROPERTIES FIXTURES_CLEANUP record_persistence)
endif()
if (EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/test.cpp")
    add_executable(file_manage_test test.cpp)
    target_link_libraries(file_manage_test PRIVATE file_manage)
    enable_testing()
    add_test(NAME file_manage_test COMMAND file_manage_test)
endif()
