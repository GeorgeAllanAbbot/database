cmake_minimum_required(VERSION 3.16)
project(database_file_manager LANGUAGES CXX)
add_library(file_manage file_manage/file_manage.cpp file_manage/buf_page_manager.cpp)
target_include_directories(file_manage PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(file_manage PUBLIC cxx_std_17)
target_compile_options(file_manage PRIVATE -Wall -Wextra -Wpedantic)
if (EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/test.cpp")
    add_executable(file_manage_test test.cpp)
    target_link_libraries(file_manage_test PRIVATE file_manage)
    enable_testing()
    add_test(NAME file_manage_test COMMAND file_manage_test)
endif()
